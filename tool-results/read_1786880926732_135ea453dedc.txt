    37→function StepIndicator({ current, steps }: { current: WizardStep; steps: typeof STEPS }) {
    38→  const currentIndex = steps.findIndex(s => s.key === current);
    39→
    40→  return (
    41→    <div className="flex items-center gap-1 overflow-x-auto pb-2 mb-4">
    42→      {steps.map((step, idx) => {
    43→        const isCompleted = idx < currentIndex;
    44→        const isCurrent = idx === currentIndex;
    45→        const isActive = isCompleted || isCurrent;
    46→
    47→        return (
    48→          <React.Fragment key={step.key}>
    49→            {idx > 0 && (
    50→              <div className={`w-4 h-0.5 ${isActive ? 'bg-neon' : 'bg-muted'}`} />
    51→            )}
    52→            <div
    53→              className={`flex items-center gap-1 px-2 py-1 rounded-full text-xs whitespace-nowrap transition-all ${
    54→                isCurrent
    55→                  ? 'shocking-text-sm bg-shocking/10 border border-shocking'
    56→                  : isCompleted
    57→                  ? 'neon-text-sm bg-neon/10 border border-neon'
    58→                  : 'text-muted-foreground bg-surface border border-border'
    59→              }`}
    60→            >
    61→              {step.icon}
    62→              <span className="hidden sm:inline">{step.label}</span>
    63→            </div>
    64→          </React.Fragment>
    65→        );
    66→      })}
    67→    </div>
    68→  );
    69→}

   165→function PlayerNamesStep() {
   166→  const { setup, updateSetup, setWizardStep } = useTournamentStore();
   167→  const [error, setError] = useState('');
   168→
   169→  const handleNameChange = (index: number, name: string) => {
   170→    const newPlayers = [...setup.players];
   171→    newPlayers[index] = { ...newPlayers[index], name };
   172→    updateSetup({ players: newPlayers });
   173→  };
   174→
   175→  const handleGenderChange = (index: number, gender: 'M' | 'F') => {
   176→    const newPlayers = [...setup.players];
   177→    newPlayers[index] = { ...newPlayers[index], gender };
   178→    updateSetup({ players: newPlayers });
   179→  };
   180→
   181→  const handleAutoName = () => {
   182→    const newPlayers = setup.players.map((_, i) => ({
   183→      name: `Giocatore ${i + 1}`,
   184→      gender: (i % 2 === 0 ? 'M' : 'F') as 'M' | 'F',
   185→    }));
   186→    updateSetup({ players: newPlayers });
   187→  };
   188→
   189→  const handleNext = () => {
   190→    const emptyNames = setup.players.filter((p) => !p.name.trim());
   191→    if (emptyNames.length > 0) {
   192→      setError('Tutti i giocatori devono avere un nome');
   193→      return;
   194→    }
   195→    const duplicateNames = setup.players.filter(
   196→      (p, i) => setup.players.findIndex((q) => q.name.trim().toLowerCase() === p.name.trim().toLowerCase()) !== i
   197→    );
   198→    if (duplicateNames.length > 0) {
   199→      setError('I nomi devono essere univoci');
   200→      return;
   201→    }
   202→    setError('');
   203→    setWizardStep('mixed');
   204→  };
   205→
   206→  return (
   207→    <div className="space-y-4">
   208→      <div className="flex items-center justify-between">
   209→        <div>
   210→          <h2 className="text-xl font-bold neon-text">Inserisci i giocatori</h2>
   211→          <p className="text-neon-dim text-sm">Nome e sesso per ogni partecipante</p>
   212→        </div>
   213→        <Button variant="outline" className="text-neon-dim border-border text-xs" onClick={handleAutoName}>
   214→          <Zap className="w-3 h-3 mr-1" /> Auto
   215→        </Button>
   216→      </div>
   217→
   218→      <div className="max-h-[50vh] overflow-y-auto space-y-2 pr-2">
   219→        {setup.players.map((player, idx) => (
   220→          <div key={idx} className="flex items-center gap-2">
   221→            <span className="text-neon-dim text-sm w-8 text-right shrink-0">#{idx + 1}</span>
   222→            <Input
   223→              value={player.name}
   224→              onChange={(e) => handleNameChange(idx, e.target.value)}
   225→              placeholder="Nome..."
   226→              className="flex-1 bg-surface text-neon border-border focus:border-neon"
   227→            />
   228→            <div className="flex gap-1">
   229→              <Button
   230→                size="sm"
   231→                variant={player.gender === 'M' ? 'default' : 'outline'}
   232→                className={
   233→                  player.gender === 'M'
   234→                    ? 'bg-shocking text-white text-xs px-3'
   235→                    : 'text-neon-dim border-border text-xs px-3 hover:border-neon'
   236→                }
   237→                onClick={() => handleGenderChange(idx, 'M')}

   239→                ♂

   242→                size="sm"

   245→                  player.gender === 'F'

   251→                ♀

   254→          </div>

   266→          variant="outline"
   267→          className="text-neon-dim border-border"
   268→          onClick={() => setWizardStep('num-players')}

   273→          onClick={handleNext}

   412→function DefinePairsStep() {
   413→  const { setup, updateSetup, setWizardStep } = useTournamentStore();
   414→  const [error, setError] = useState('');
   415→
   416→  const males = setup.players.map((p, i) => ({ ...p, index: i })).filter((p) => p.gender === 'M');
   417→  const females = setup.players.map((p, i) => ({ ...p, index: i })).filter((p) => p.gender === 'F');
   418→
   419→  const availableMales = males.filter(
   420→    (m) => !setup.fixedPairs.some((fp) => fp.player1Index === m.index || fp.player2Index === m.index)
   421→  );
   422→  const availableFemales = females.filter(
   423→    (f) => !setup.fixedPairs.some((fp) => fp.player1Index === f.index || fp.player2Index === f.index)
   424→  );
   425→
   426→  const totalPairs = Math.floor(setup.players.length / 2);
   427→
   428→  const addPair = () => {
   429→    if (setup.fixedPairs.length >= totalPairs) return;
   430→
   431→    if (setup.isMixed) {
   432→      if (availableMales.length === 0 || availableFemales.length === 0) return;
   433→      updateSetup({
   434→        fixedPairs: [
   435→          ...setup.fixedPairs,
   436→          { player1Index: availableMales[0].index, player2Index: availableFemales[0].index },
   437→        ],
   438→      });
   439→    } else {
   440→      const allAvailable = setup.players
   441→        .map((p, i) => ({ ...p, index: i }))
   442→        .filter(
   443→          (p) => !setup.fixedPairs.some((fp) => fp.player1Index === p.index || fp.player2Index === p.index)
   444→        );
   445→      if (allAvailable.length < 2) return;
   446→      updateSetup({
   447→        fixedPairs: [
   448→          ...setup.fixedPairs,
   449→          { player1Index: allAvailable[0].index, player2Index: allAvailable[1].index },
   450→        ],
   451→      });
   452→    }
   453→  };
   454→
   455→  const removePair = (index: number) => {
   456→    updateSetup({
   457→      fixedPairs: setup.fixedPairs.filter((_, i) => i !== index),
   458→    });
   459→  };
   460→
   461→  const handlePlayerChange = (pairIndex: number, position: 'player1' | 'player2', newPlayerIndex: number) => {
   462→    const newPairs = [...setup.fixedPairs];
   463→    newPairs[pairIndex] = { ...newPairs[pairIndex], [`${position}Index`]: newPlayerIndex };
   464→    updateSetup({ fixedPairs: newPairs });
   465→  };
   466→
   467→  const handleAutoPair = () => {
   468→    const pairs: { player1Index: number; player2Index: number }[] = [];
   469→
   470→    if (setup.isMixed) {
   471→      const availM = [...males];
   472→      const availF = [...females];
   473→      const numPairs = Math.min(availM.length, availF.length);
   474→      for (let i = 0; i < numPairs; i++) {
   475→        pairs.push({ player1Index: availM[i].index, player2Index: availF[i].index });
   476→      }
   477→    } else {
   478→      const allPlayers = setup.players.map((p, i) => ({ ...p, index: i }));
   479→      for (let i = 0; i + 1 < allPlayers.length; i += 2) {
   480→        pairs.push({ player1Index: allPlayers[i].index, player2Index: allPlayers[i + 1].index });
   481→      }
   482→    }
   483→
   484→    updateSetup({ fixedPairs: pairs });
   485→  };
   486→
   487→  const handleNext = () => {
   488→    const pairedPlayers = new Set<number>();
   489→    for (const fp of setup.fixedPairs) {
   490→      pairedPlayers.add(fp.player1Index);
   491→      pairedPlayers.add(fp.player2Index);
   492→    }
   493→
   494→    const unpaired = setup.players.length - pairedPlayers.size;
   495→    if (unpaired > 1) {
   496→      setError(`Ci sono ${unpaired} giocatori senza coppia. Assegnali o rimuovili.`);
   497→      return;
   498→    }
   499→
   500→    if (setup.fixedPairs.length < 2) {
   501→      setError('Servono almeno 2 coppie');
   502→      return;
   503→    }
   504→
   505→    setError('');
   506→    setWizardStep('courts-days');
   507→  };
   508→
   509→  return (
   510→    <div className="space-y-4">
   511→      <div className="flex items-center justify-between">
   512→        <div>
   513→          <h2 className="text-xl font-bold neon-text">Definisci le Coppie</h2>
   514→          <p className="text-neon-dim text-sm">
   515→            {setup.isMixed ? 'Accoppia ogni uomo con una donna' : 'Forma le coppie'}
   516→          </p>
   517→        </div>
   518→        <Button variant="outline" className="text-neon-dim border-border text-xs" onClick={handleAutoPair}>
   519→          <Zap className="w-3 h-3 mr-1" /> Auto
   520→        </Button>
   521→      </div>
   522→
   523→      <div className="max-h-[45vh] overflow-y-auto space-y-2 pr-2">
   524→        {setup.fixedPairs.map((pair, idx) => {
   525→          const p1 = setup.players[pair.player1Index];
   526→          const p2 = setup.players[pair.player2Index];
   527→
   528→          return (
   529→            <div key={idx} className="flex items-center gap-2 bg-surface p-2 rounded border border-border">
   530→              <Badge className="bg-neon/10 text-neon border border-neon">Coppia {idx + 1}</Badge>
   531→              <div className="flex-1 flex items-center gap-2">
   532→                <select
   533→                  value={pair.player1Index}
   534→                  onChange={(e) => handlePlayerChange(idx, 'player1', parseInt(e.target.value))}
   535→                  className="flex-1 bg-surface text-neon border border-border rounded px-2 py-1 text-sm"
   536→                >
   537→                  {setup.players.map((p, i) => (
   538→                    <option key={i} value={i} className="bg-surface text-neon">
   539→                      {p.name} ({p.gender === 'M' ? '♂' : '♀'})
   540→                    </option>
